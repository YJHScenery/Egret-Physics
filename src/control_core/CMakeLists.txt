# /** @file CMakeLists.txt @brief 无 Qt 编译执行核心与独立验证宿主。 */
cmake_minimum_required(VERSION 3.30)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(EgretControlCore LANGUAGES CXX)
    enable_testing()
    option(BUILD_TESTS "构建控制核心验证宿主" ON)
endif()
set(EGRET_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../..")
set(EGRET_CONTROL_DEPS "${EGRET_ROOT}/dependency/control" CACHE PATH "场景控制依赖目录")
if(NOT EXISTS "${EGRET_CONTROL_DEPS}/lua-5.4.8/src/lua.h")
    message(FATAL_ERROR "缺少 Lua 5.4.8，请先运行 tools/setup_control_dependencies.ps1")
endif()
set(EGRET_WASMTIME_ROOT "${EGRET_CONTROL_DEPS}/wasmtime-v38.0.3-x86_64-windows-c-api")
add_library(Egret_Control_Core STATIC control_program.cpp lua_program.cpp)
set_target_properties(Egret_Control_Core PROPERTIES AUTOMOC OFF AUTOUIC OFF AUTORCC OFF)
target_compile_features(Egret_Control_Core PUBLIC cxx_std_20)
target_compile_options(Egret_Control_Core PRIVATE /utf-8 /EHsc)
target_include_directories(Egret_Control_Core PUBLIC "${EGRET_ROOT}/src" PRIVATE "${EGRET_CONTROL_DEPS}" "${EGRET_WASMTIME_ROOT}/include")
target_link_libraries(Egret_Control_Core PUBLIC "${EGRET_WASMTIME_ROOT}/lib/wasmtime.dll.lib" bcrypt)
if(BUILD_TESTS)
enable_testing()
add_executable(Egret_Programmable_Control_Tests "${EGRET_ROOT}/tests/test_programmable_control.cpp")
set_target_properties(Egret_Programmable_Control_Tests PROPERTIES AUTOMOC OFF AUTOUIC OFF AUTORCC OFF)
target_compile_options(Egret_Programmable_Control_Tests PRIVATE /utf-8 /EHsc)
target_link_libraries(Egret_Programmable_Control_Tests PRIVATE Egret_Control_Core)
add_custom_command(TARGET Egret_Programmable_Control_Tests POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_if_different "${EGRET_WASMTIME_ROOT}/lib/wasmtime.dll" "$<TARGET_FILE_DIR:Egret_Programmable_Control_Tests>")
add_test(NAME EgretProgrammableControl COMMAND Egret_Programmable_Control_Tests "${EGRET_CONTROL_DEPS}/llvm/bin/clang.exe" "${CMAKE_CURRENT_SOURCE_DIR}/sdk")
set_tests_properties(EgretProgrammableControl PROPERTIES ENVIRONMENT "WASMTIME_BACKTRACE_DETAILS=1")
endif()

# 将可信 SDK 生成为每实例成员，两个源码后端共享相同领域实现。
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/sdk/egret_runtime.c" EGRET_NATIVE_RUNTIME)
string(REGEX REPLACE "#include [^\n]*\n" "" EGRET_NATIVE_RUNTIME "${EGRET_NATIVE_RUNTIME}")
string(REPLACE "static " "" EGRET_NATIVE_RUNTIME "${EGRET_NATIVE_RUNTIME}")
string(REPLACE "__builtin_trap();" "throw std::runtime_error(\"控制 SDK 参数、句柄或容量无效\");" EGRET_NATIVE_RUNTIME "${EGRET_NATIVE_RUNTIME}")
string(REPLACE "__builtin_sqrt" "std::sqrt" EGRET_NATIVE_RUNTIME "${EGRET_NATIVE_RUNTIME}")
string(REPLACE "__builtin_fabs" "std::fabs" EGRET_NATIVE_RUNTIME "${EGRET_NATIVE_RUNTIME}")
string(REPLACE "__builtin_floor" "std::floor" EGRET_NATIVE_RUNTIME "${EGRET_NATIVE_RUNTIME}")
string(REGEX REPLACE "\\(([A-Za-z0-9_]+)\\) *\\{" "\\1{" EGRET_NATIVE_RUNTIME "${EGRET_NATIVE_RUNTIME}")
string(FIND "${EGRET_NATIVE_RUNTIME}" "unsigned egret_exchange(void)" EGRET_NATIVE_END)
string(SUBSTRING "${EGRET_NATIVE_RUNTIME}" 0 ${EGRET_NATIVE_END} EGRET_NATIVE_RUNTIME)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/native_runtime.inc" "${EGRET_NATIVE_RUNTIME}")
target_include_directories(Egret_Control_Core PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/sdk/egret_runtime.c")
# Lua 使用 C++ 异常展开，保证宿主绑定中的对象在脚本出错时正确析构。
file(GLOB EGRET_LUA_SOURCES "${EGRET_CONTROL_DEPS}/lua-5.4.8/src/*.c")
list(FILTER EGRET_LUA_SOURCES EXCLUDE REGEX "/(lua|luac)\\.c$")
set_source_files_properties(${EGRET_LUA_SOURCES} PROPERTIES LANGUAGE CXX)
add_library(Egret_Lua STATIC ${EGRET_LUA_SOURCES})
set_target_properties(Egret_Lua PROPERTIES AUTOMOC OFF AUTOUIC OFF AUTORCC OFF)
target_compile_options(Egret_Lua PRIVATE /EHsc)
target_include_directories(Egret_Lua PUBLIC "${EGRET_CONTROL_DEPS}/lua-5.4.8/src")
target_link_libraries(Egret_Control_Core PRIVATE Egret_Lua)

if(BUILD_TESTS)
    add_executable(Egret_Lua_Control_Tests "${EGRET_ROOT}/tests/test_lua_control.cpp")
    set_target_properties(Egret_Lua_Control_Tests PROPERTIES AUTOMOC OFF AUTOUIC OFF AUTORCC OFF)
    target_compile_options(Egret_Lua_Control_Tests PRIVATE /utf-8 /EHsc)
    target_link_libraries(Egret_Lua_Control_Tests PRIVATE Egret_Control_Core)
    add_custom_command(TARGET Egret_Lua_Control_Tests POST_BUILD
        COMMAND ${CMAKE_COMMAND} -E copy_if_different "${EGRET_WASMTIME_ROOT}/lib/wasmtime.dll" "$<TARGET_FILE_DIR:Egret_Lua_Control_Tests>")
    add_test(NAME EgretLuaControl COMMAND Egret_Lua_Control_Tests "${EGRET_CONTROL_DEPS}/llvm/bin/clang.exe" "${CMAKE_CURRENT_SOURCE_DIR}/sdk")
endif()

# 独立核心及其验证宿主也必须同步共享结构，不能依赖不完整的包含扫描。
include("${EGRET_ROOT}/cmake/ControlDependencies.cmake")
egret_apply_control_dependencies(Egret_Control_Core)
if(BUILD_TESTS)
    egret_apply_control_dependencies(Egret_Programmable_Control_Tests)
    egret_apply_control_dependencies(Egret_Lua_Control_Tests)
endif()


# 数学实现随 SDK 分发；用户代码编译阶段仅链接预编译归档。
if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/sdk/libegret_math.a")
    message(FATAL_ERROR "缺少预编译数学库，请运行 tools/setup_control_math.ps1")
endif()
